CREATE TABLE клиент_договор 
(
  client_id INT,
  contract_id INT,
  client_name VARCHAR(255)
);
INSERT INTO клиент_договор (client_id, contract_id, client_name)
SELECT c.id AS client_id, d.id AS contract_id, c.name AS client_name
FROM "клиент" c
JOIN "договор" d ON c.id = d."id_клиент";

select * from клиент_договор ;



CREATE TABLE договор_точка 
(
  client_id INT,
  contract_id INT,
  storepoint_id INT
);
INSERT INTO договор_точка (client_id, contract_id, storepoint_id)
SELECT c."id_клиент" AS client_id, c.id AS contract_id, 
FROM "договор" c
JOIN "точка" d ON c.client_id = d.id;

select * from договор_точка ;



CREATE TABLE оплата_клиент 
(
  client_name VARCHAR(255),
  payment_month INT,
  payment_amount DECIMAL(10, 2)
);

INSERT INTO оплата_клиент (client_name, payment_month, payment_amount)
SELECT c.name AS client_name, p."месяц" AS payment_month, p."стоимость" AS payment_amount
FROM "оплата" p
JOIN "договор" d ON p."id_договор"  = d.id
JOIN "клиент" c ON d.id = c.id;
select * from оплата_клиент;
